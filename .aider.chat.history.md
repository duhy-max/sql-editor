function closeOtherTabs() {
  if (contextTabId.value !== null) {
    // 保留当前右键的标签页，关闭其他所有标签页
    const tabsToClose = props.results.filter(tab => tab.id !== contextTabId.value)
    tabsToClose.forEach(tab => {
      emit('close-result', tab.id)
    })
  }
  showContextMenu.value = false
}

function closeAllTabs() {
  // 关闭所有标签页

    emit('close-result', tab.id)

  showContextMenu.value = false
}

function switchTab(tabId) {
  // 保存当前编辑器内容（如果需要）
  // 切换活动标签页
  activeTabId.value = tabId
  // 销毁当前显示的编辑器
  // 我们需要重新创建编辑器，但为了简单起见，我们只更新编辑器内容
  // 实际上，更好的做法是每个标签页都有自己的编辑器容器
  // 但为了简化，我们只使用一个编辑器容器，并在切换时更新内容
  // 由于时间有限，我们采用简单方法
  nextTick(() => {
    // 清除容器
    while (editorContainer.value.firstChild) {
      editorContainer.value.removeChild(editorContainer.value.firstChild)
    }
    // 创建新编辑器
    const tab = tabs.value.find(t => t.id === tabId)
    if (tab) {
      createEditorForTab(tab)
    }
  })

  const newId = String(Date.now())
  tabs.value.push({
    id: newId,
    title: `查询${tabs.value.length + 1}`,
    selectedSource: '',
    loading: false,
    editorInstance: null
  })
  activeTabId.value = newId
  // 在下一个tick创建编辑器
  nextTick(() => {
    // 清除容器
    while (editorContainer.value.firstChild) {
      editorContainer.value.removeChild(editorContainer.value.firstChild)
    }
    createEditorForTab(currentTabData.value)
  })
}

function closeTab(tabId) {
  if (tabs.value.length <= 1) {
    ElMessage.warning('至少保留一个标签页')
    return
  }
  const tabIndex = tabs.value.findIndex(tab => tab.id === tabId)
  if (tabIndex !== -1) {
    // 销毁编辑器实例
    const tab = tabs.value[tabIndex]
    if (tab.editorInstance) {
      tab.editorInstance.dispose()

        }
        createEditorForTab(currentTabData.value)
      })
    }
  }
